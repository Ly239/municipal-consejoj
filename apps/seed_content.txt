                'is_active': True,
            },
            {

Este modelo comenzó a transformarse con la Constitución de 1999 y se consolidó definitivamente con la Ley Orgánica del Poder Público Municipal del año 2005[cite: 10].\n\nDicho hito jurídico consagró la Separación de Poderes a nivel local, transformando al Concejo en un Cuerpo Edilicio Autónomo e independiente.                 'category': cat_historia,
                'is_active': True,
            },
            {
                'title': 'Nuestros Valores',
                'slug': slugify('Nuestros Valores'),
                'summary': 'Transparencia, Responsabilidad, Institucionalidad, Participación, Ética Pública y Compromiso Social.',
                'content': '- Transparencia: Garantizar el acceso público a la información y el manejo pulcro de los recursos, permitiendo que cada ciudadano sea testigo de la gestión legislativa[cite: 4].\n- Responsabilidad: Asumir con integridad el compromiso de legislar y fiscalizar, respondiendo con eficacia a las necesidades del Municipio Junín[cite: 4].\n- Institucionalidad: Actuar con estricto apego al ordenamiento jurídico y al respeto de las competencias de cada poder[cite: 4].\n- Participación: Promover mecanismos para que el ciudadano sea protagonista activo en la toma de decisiones[cite: 4].\n- Ética Pública: Mantener una conducta intachable y honesta en el ejercicio de la función pública[cite: 4].\n- Compromiso Social: Orientar la labor legislativa hacia la construcción del bienestar común de los habitantes de la Ciudad de los Puentes[cite: 4].',

            }

        for item in about_data:
            AboutUs.objects.update_or_create(slug=item['slug'], defaults=item)
            self.stdout.write(self.style.SUCCESS(f'  ✔ [AboutUs] Cargado: {item["title"]}'))

            {'name': 'Danny Sepulveda', 'charge': 'Concejal', 'bio': 'Concejal principal del municipio[cite: 11].'},

                    'name': c['name'],
                    'charge': c['charge'],
                    'bio': c['bio'],
                    'is_active': True
                }

            "Transporte",
            "Salud, Ambiente y Actividades Agrícolas y Pecuarias",
            "Comunicación Social y Tecnología",
            "Eventos Públicos"
        ]

                defaults={
                    'description': f'Comisión Legislativa Permanente N° {idx} del Concejo Municipal de Junín[cite: 12, 13].',
                    'is_active': True
                }
            )
            self.stdout.write(self.style.SUCCESS(f'  ✔ [Commission] Creada: {nombre_comision}'))

        self.stdout.write(self.style.SUCCESS('🎉 ¡Todos los datos institucionales reales se han cargado e inyectado con éxito!'))
